Label

Form labels.

import { Label } from "@/components/ui/label";
import { Textarea } from "@/components/ui/textarea";

export default function LabelExample() {
  return (
    <div className="grid w-full max-w-sm gap-2">
      <Label htmlFor="note">Note for the team</Label>
      <Textarea id="note" placeholder="What changed and why" />
    </div>
  );
}

Installation

npx auto-design add {origin}/r/ --components label
npx shadcn@latest add {origin}/r/label.json

Install the dependencies, then copy the source into components/ui/label.tsx.

npm install radix-ui 
"use client";

import * as React from "react";
import { Label as LabelPrimitive } from "radix-ui";

import { cn } from "@/lib/utils";

function Label({ className, ...props }: React.ComponentProps<typeof LabelPrimitive.Root>) {
  return (
    <LabelPrimitive.Root
      data-slot="label"
      className={cn("flex select-none items-center gap-2 text-sm font-medium leading-none peer-disabled:cursor-not-allowed peer-disabled:opacity-50", className)}
      {...props}
    />
  );
}

export { Label };

Usage

import { Label } from "@/components/ui/label";

<div className="grid w-full max-w-sm gap-2">
  <Label htmlFor="note">Note for the team</Label>
  <Textarea id="note" placeholder="What changed and why" />
</div>
