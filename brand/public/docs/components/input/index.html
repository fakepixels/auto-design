Input

Text input sized to match buttons.

Shown to everyone on the team.

import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";

export default function InputExample() {
  return (
    <div className="grid w-full max-w-sm gap-2">
      <Label htmlFor="project-name">Project name</Label>
      <Input id="project-name" placeholder="Website refresh" />
      <p className="text-sm text-muted-foreground">Shown to everyone on the team.</p>
    </div>
  );
}

Installation

npx auto-design add {origin}/r/ --components input
npx shadcn@latest add {origin}/r/input.json

Copy the source into components/ui/input.tsx.

import * as React from "react";

import { cn } from "@/lib/utils";

function Input({ className, type, ...props }: React.ComponentProps<"input">) {
  return (
    <input
      type={type}
      data-slot="input"
      className={cn(
        "flex h-9 w-full min-w-0 rounded-md border border-input bg-background px-3 text-base shadow-xs transition-colors duration-120 ease-out placeholder:text-muted-foreground focus-visible:border-ring disabled:cursor-not-allowed disabled:opacity-50 aria-invalid:border-destructive file:border-0 file:bg-transparent file:text-sm file:font-medium",
        className,
      )}
      {...props}
    />
  );
}

export { Input };
